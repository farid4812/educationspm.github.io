Hardware Wiring • ESP32-C5 + CNC Shield V3 (A4988)

⚠ NEVER CONNECT 12V–24V VMOT DIRECTLY TO ESP32
1. CNC Shield V3 Power

12V–24V DC connects to CNC Shield V3 blue terminal (+/−) with 100µF decoupling capacitors under each A4988 module.

2. A4988 & Microstepping

Install all 3 jumpers (MS1, MS2, MS3) under X & Y A4988 sockets for 1/16 microstepping. Tune VREF pot to ~0.65V.

3. Shared EN & Servo

CNC Shield EN pin (active LOW) enables all A4988 drivers simultaneously. Pen Servo connects to Z+ / Spindle PWM header.

ESP32-C5 + CNC SHIELD V3 (A4988) FIRMWARE 
#include <AccelStepper.h>
#include <ESP32Servo.h>
#include <DHT.h>

// ESP32-C5 -> CNC Shield V3 (A4988 Stepper Drivers) Pin Map
#define X_STEP_PIN  2   // Shield D2
#define X_DIR_PIN   3   // Shield D5
#define Y_STEP_PIN  4   // Shield D3
#define Y_DIR_PIN   5   // Shield D6
#define SHIELD_EN   11  // Shield D8 (Active LOW enables A4988)
#define SERVO_PIN   6   // Shield Z+ / SpnEn PWM
#define DHT_PIN     7
#define LED_GREEN   8
#define LED_YELLOW  9
#define LED_RED     10

const float STEPS_PER_MM = 80.0; // 200 steps * 16 microsteps / 40mm belt
AccelStepper stepperX(AccelStepper::DRIVER, X_STEP_PIN, X_DIR_PIN);
AccelStepper stepperY(AccelStepper::DRIVER, Y_STEP_PIN, Y_DIR_PIN);
Servo penServo;
DHT dht(DHT_PIN, DHT22);

void penUp()   { penServo.write(90); delay(100); } // M5
void penDown() { penServo.write(25); delay(100); } // M3

void setup() {
  Serial.begin(115200);
  pinMode(SHIELD_EN, OUTPUT);
  digitalWrite(SHIELD_EN, LOW); // Enable A4988 drivers
  dht.begin();
  penServo.attach(SERVO_PIN);
  penUp();
  stepperX.setMaxSpeed(2400); stepperX.setAcceleration(1200);
  stepperY.setMaxSpeed(2400); stepperY.setAcceleration(1200);
  pinMode(LED_GREEN, OUTPUT); pinMode(LED_YELLOW, OUTPUT); pinMode(LED_RED, OUTPUT);
  Serial.println("ESP32-C5 CNC Shield V3 (A4988) Ready");
}

void loop() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "M5") { penUp(); Serial.println("ok"); }
    else if (cmd == "M3") { penDown(); Serial.println("ok"); }
    else if (cmd == "M17") { digitalWrite(SHIELD_EN, LOW); Serial.println("ok"); }
    else if (cmd == "M18") { digitalWrite(SHIELD_EN, HIGH); Serial.println("ok"); }
    else if (cmd == "G28") {
      penUp();
      stepperX.moveTo(18 * STEPS_PER_MM);
      stepperY.moveTo(18 * STEPS_PER_MM);
      while (stepperX.distanceToGo() != 0 || stepperY.distanceToGo() != 0) {
        stepperX.run(); stepperY.run();
      }
      Serial.println("ok");
    }
    else if (cmd.startsWith("G0") || cmd.startsWith("G1")) {
      int xIdx = cmd.indexOf('X'), yIdx = cmd.indexOf('Y');
      if (xIdx >= 0) stepperX.moveTo(cmd.substring(xIdx + 1).toFloat() * STEPS_PER_MM);
      if (yIdx >= 0) stepperY.moveTo(cmd.substring(yIdx + 1).toFloat() * STEPS_PER_MM);
      while (stepperX.distanceToGo() != 0 || stepperY.distanceToGo() != 0) {
        stepperX.run(); stepperY.run();
      }
      Serial.println("ok");
    }
  }
  float temp = dht.readTemperature();
  if (!isnan(temp)) {
    digitalWrite(LED_GREEN,  temp < 30.0);
    digitalWrite(LED_YELLOW, temp >= 30.0 && temp <= 35.0);
    digitalWrite(LED_RED,    temp > 35.0);
  }
}
